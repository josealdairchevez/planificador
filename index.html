const functions = require('firebase-functions'); const admin = require('firebase-admin'); admin.initializeApp(); const db = admin.firestore(); const messaging = admin.messaging(); // ── Se ejecuta cada minuto ────────────────────────────────── exports.sendScheduledReminders = functions.pubsub .schedule('every 1 minutes') .onRun(async (context) => { const now = Date.now(); // ── CORRECCIÓN 1: Ventana de tiempo estricta ── const windowEnd = now + 15000; // Solo 15 segundos hacia el futuro const windowStart = now - 60000; // Hasta 60 segundos en el pasado console.log(`Revisando recordatorios entre ${new Date(windowStart).toISOString()} y ${new Date(windowEnd).toISOString()}`); try { const usersSnap = await db.collection('users').get(); const promises = []; usersSnap.forEach(userDoc => { const data = userDoc.data(); const reminders = data.fcm_reminders || []; let tokensToUse = []; if (Array.isArray(data.fcm_tokens) && data.fcm_tokens.length > 0) { tokensToUse = data.fcm_tokens; } else if (data.fcm_token) { tokensToUse = [data.fcm_token]; } if (tokensToUse.length === 0 || !reminders.length) return; const toFire = reminders.filter(r => r.fireAt >= windowStart && r.fireAt <= windowEnd); let remaining = reminders.filter(r => r.fireAt > windowEnd); if (toFire.length === 0) return; console.log(`Usuario ${userDoc.id}: disparando ${toFire.length} recordatorio(s)`); toFire.forEach(r => { const titleText = String(r.title || 'Recordatorio'); const bodyText = String(r.body || 'Tienes un recordatorio'); tokensToUse.forEach(fcmToken => { // ── CORRECCIÓN 2: Paquete "Data-Only" para evitar notificaciones dobles ── const message = { token: fcmToken, data: { title: titleText, body: bodyText, evId: String(r.evId || ''), fireAt: String(r.fireAt || '') }, android: { priority: 'high' }, webpush: { headers: { Urgency: 'high' } } }; promises.push( messaging.send(message).catch(err => { console.error(`Error enviando a usuario ${userDoc.id}:`, err.message); if (err.code === 'messaging/invalid-registration-token' || err.code === 'messaging/registration-token-not-registered') { return db.collection('users').doc(userDoc.id) .update({ fcm_token: null, fcm_tokens: admin.firestore.FieldValue.arrayRemove(fcmToken) }); } }) ); }); // ── Reprogramación de hábitos para el día siguiente ── if (r.evId && String(r.evId).startsWith('hab_')) { const nextFireAt = (r.fireAt || now) + 86400000; // +24h remaining.push({ evId: r.evId, fireAt: nextFireAt, title: r.title, body: r.body }); console.log(`Hábito reprogramado para ${new Date(nextFireAt).toISOString()}: ${r.title}`); } }); promises.push( db.collection('users').doc(userDoc.id) .update({ fcm_reminders: remaining }) ); }); await Promise.all(promises); console.log('Ciclo completado.'); } catch (err) { console.error('Error en sendScheduledReminders:', err); } return null; });
